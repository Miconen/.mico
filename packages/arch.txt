# Packages pacman owns on the Arch laptop only.
#
# Everything here is hardware, desktop, or GUI - the categories nix explicitly
# does not manage in this setup.

# CPU microcode
amd-ucode

# Boot
efibootmgr

# GPU / video acceleration.
# This laptop is an AMD Carrizo APU (Wani, Radeon R5/R6/R7, GCN 3) driven by
# amdgpu - confirmed via lspci. So no Intel and no Nouveau packages here.
#
# mesa is listed explicitly because it is now the VA-API provider: it
# `provides libva-driver` and `replaces libva-mesa-driver`, which was merged
# into mesa in 24.2.7. Before this, the only VA-API drivers installed were the
# Intel ones, meaning hardware video decode did nothing on this machine.
mesa
vulkan-radeon

# Xorg DDX driver. Arguably unnecessary under Hyprland/Wayland, and Xorg's
# built-in modesetting driver handles amdgpu fine - but sddm can still start on
# X11, and "probably unnecessary" is not worth risking a machine that cannot
# reach a desktop. Safe to drop if you confirm sddm runs on Wayland.
xf86-video-amdgpu

# Wayland compositor and session
hyprland
uwsm
xdg-desktop-portal-hyprland
qt5-wayland
qt6-wayland
xdg-utils

# Desktop environment / login
plasma-meta
sddm
polkit-kde-agent
dolphin

# Terminal emulator. Config is managed by nix in hosts/arch.nix.
kitty

# Launcher, notifications, screenshots
wofi
dunst
grim
slurp

# Audio
pipewire
pipewire-alsa
pipewire-jack
pipewire-pulse
wireplumber
gst-plugin-pipewire
libpulse

# Network
networkmanager
network-manager-applet
wpa_supplicant

# Bluetooth
bluez
bluez-utils

# Printing
cups
cups-pk-helper
system-config-printer

# Hardware / misc
smartmontools
zram-generator
fuse2

# Browser
firefox

# AUR helper
paru

# Still used to manage nothing in particular now that dotfiles moved to nix,
# but harmless to keep around while the migration settles.
stow
